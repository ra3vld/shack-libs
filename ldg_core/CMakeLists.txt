set(ldg_dir "${CMAKE_CURRENT_LIST_DIR}")
set(ldg_sources "${ldg_dir}/src/protocol.cpp" "${ldg_dir}/src/controller.cpp")
if(COMMAND idf_component_register)
  idf_component_register(SRCS ${ldg_sources} INCLUDE_DIRS "${ldg_dir}/include" REQUIRES arduinojson)
  target_compile_features(${COMPONENT_LIB} PUBLIC cxx_std_20)
else()
  add_library(ldg_core STATIC ${ldg_sources})
  target_include_directories(ldg_core PUBLIC "${ldg_dir}/include")
  target_link_libraries(ldg_core PUBLIC arduinojson)
  target_compile_features(ldg_core PUBLIC cxx_std_20)
endif()
